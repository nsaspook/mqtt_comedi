147void led_lightshow(int speed)
148{
149 static int j = 0;
150 static uint8_t cylon = 0xff;
151 static int alive_led = 0;
152 static bool LED_UP = true;
153
154 if (j++ >= speed) { // delay a bit ok
155 if (0) { // screen status feedback
156 bmc.dataout.dio_buf = ~cylon; // roll leds cylon style
157 } else {
158 bmc.dataout.dio_buf = cylon; // roll leds cylon style (inverted)
159 }
160
161 if (LED_UP && (alive_led != 0)) {
162 alive_led = alive_led * 2;
163 cylon = cylon << 1;
164 } else {
165 if (alive_led != 0) alive_led = alive_led / 2;
166 cylon = cylon >> 1;
167 }
168 if (alive_led < 2) {
169 alive_led = 2;
170 LED_UP = true;
171 } else {
172 if (alive_led > 128) {
173 alive_led = 128;
174 LED_UP = false;
175 }
176 }
177 j = 0;
178 }
179}

192bool get_set_config(void)
193{
194 static const char *output_file = "/etc/daq_bmc/bmc_config.cfg"; // will only read data from here
195 static const char *output_file_tmp = "/tmp/bmc_config.cfg"; // if the correct directory is missing
196 const char *tmp_mqtt;
197 config_t cfg;
198 config_setting_t *setting, *group;
199 bool config_status = false;
200
201 /*
202 * read configuration file settings
203 */
204 config_init(&cfg);
205 config_set_options(&cfg,
206#ifdef RPIDAQ
207 (CONFIG_OPTION_FSYNC
208 | CONFIG_OPTION_SEMICOLON_SEPARATORS
209#else
210 (CONFIG_OPTION_SEMICOLON_SEPARATORS
211#endif
212 | CONFIG_OPTION_COLON_ASSIGNMENT_FOR_GROUPS
213 | CONFIG_OPTION_OPEN_BRACE_ON_SEPARATE_LINE));
214 /* Read the file. If there is an error, create a new file */
215 if (!config_read_file(&cfg, output_file)) {
216#ifdef CFG_DEBUG
217 fprintf(stderr, "%s:%d - %s\n", config_error_file(&cfg),
218 config_error_line(&cfg), config_error_text(&cfg));
219#else
220 fprintf(stderr, "%s%s\n", (char*) "Unable to open BMC configuration file ", (char*) output_file);
221#endif
222 config_destroy(&cfg);
223
224 config_init(&cfg);
225 group = config_root_setting(&cfg);
226 setting = config_setting_add(group, "BENERGYV", CONFIG_TYPE_FLOAT);
227 config_setting_set_float(setting, S.BENERGYV);
228 setting = config_setting_add(group, "BVOLTAGEV", CONFIG_TYPE_FLOAT);
229 config_setting_set_float(setting, S.BVOLTAGEV);
230 setting = config_setting_add(group, "BVFLOATV", CONFIG_TYPE_FLOAT);
231 config_setting_set_float(setting, S.BVFLOATV);
232 setting = config_setting_add(group, "PVENERGYV", CONFIG_TYPE_FLOAT);
233 config_setting_set_float(setting, S.PVENERGYV);
234 setting = config_setting_add(group, "PVVOLTAGEV", CONFIG_TYPE_FLOAT);
235 config_setting_set_float(setting, S.PVVOLTAGEV);
236 setting = config_setting_add(group, "SOC_MODEV", CONFIG_TYPE_FLOAT);
237 config_setting_set_float(setting, S.SOC_MODEV);
238 setting = config_setting_add(group, MQTT_HOST_STR, CONFIG_TYPE_STRING);
239 config_setting_set_string(setting, MQTT_HOST);
240
241 /* Write out the new configuration. */
242 if (!config_write_file(&cfg, output_file)) {
243 if (!config_write_file(&cfg, output_file_tmp)) {
244 fprintf(stderr, "Error while writing file to %s or %s.\n", output_file, output_file_tmp);
245#ifdef EXIT_CONFIG_WRITE_FAIL
246 config_destroy(&cfg);
247 return(EXIT_FAILURE);
248#endif
249 } else {
250 fprintf(stderr, "Default configuration successfully written to: %s\n",
251 output_file_tmp);
252 fprintf(stderr, "review for correctness of all parameters and check the MQTT host %s\n",
253 (char*) MQTT_HOST_STR);
254 }
255 } else {
256 fprintf(stderr, "New configuration successfully written to: %s\n",
257 output_file);
258 config_status = true;
259 }
260 config_destroy(&cfg);
261 } else {
262 if (config_lookup_float(&cfg, "BENERGYV", &S.BENERGYV)
263 && config_lookup_float(&cfg, "BVOLTAGEV", &S.BVOLTAGEV)
264 && config_lookup_float(&cfg, "BVFLOATV", &S.BVFLOATV)
265 && config_lookup_float(&cfg, "PVENERGYV", &S.PVENERGYV)
266 && config_lookup_float(&cfg, "PVVOLTAGEV", &S.PVVOLTAGEV)
267 && config_lookup_float(&cfg, "SOC_MODEV", &S.SOC_MODEV)
268 && config_lookup_string(&cfg, MQTT_HOST_STR, &tmp_mqtt)) {
269 fprintf(stderr, "Configuration successfully read: %4.1f, %4.1f, %4.1f, %4.1f, %4.1f, %4.1f, %s from: %s\n",
270 S.BENERGYV, S.BVOLTAGEV, S.BVFLOATV, S.PVENERGYV, S.PVVOLTAGEV, S.SOC_MODEV, tmp_mqtt, output_file);
271 strncpy(S.MQTT_HOSTIP, tmp_mqtt, BMC_MAXHOST - 1);
272 config_status = true;
273 } else {
274 fprintf(stderr, "No/Incorrect settings in configuration file, please review file contents.\n");
275 }
276 config_destroy(&cfg);
277 }
278 return config_status;
279}

281int main(int argc, char *argv[])
282{
283 int do_ao_only = false;
284 uint32_t i = 0, j = 75;
285
286
287 /*
288 * read configuration file settings
289 */
290 get_set_config();
291
292 /*
293 * start the MQTT processing
294 */
295 bmc_mqtt_init();
296
297 if (do_ao_only) {
298 if (init_dac(0.0, 25.0, false) < 0) {
299 fprintf(fout, "Missing Analog AO subdevice\n");
300 return -1;
301 }
302
303 while (true) {
304 set_dac_raw(0, sine_wave[255 - i++] << 4);
305 set_dac_raw(1, sine_wave[255 - j++] << 4);
306 }
307 } else {
308#ifndef DIGITAL_ONLY
309 if (init_daq(0.0, 25.0, false) < 0) {
310 fprintf(fout, "Missing Analog subdevice(s)\n");
311 return -1;
312 }
313#endif
314 if (init_dio() < 0) {
315 fprintf(fout, "Missing Digital subdevice(s)\n");
316 }
317
318 E.dac[0] = 1.23f;
319 E.dac[1] = 3.21f;
320
321 E.do_16b = 0x01;
322 E.di_16b = 0x10;
323
324 fflush(fout);
325 while (true) {
326 usleep(MAIN_DELAY); // sample rate ~1 msec
327 // set_dac_raw(0, sine_wave[i++]);
328 get_data_sample();
329
330 if (bmc.BOARD == bmcboard) {
331#ifdef USE_RELAYS
332 relay_set(0b01010101);
333#else
334 led_lightshow(10);
335#endif
336 } else {
337 if (ha_daq_host.hindex >= 4) {
338 led_lightshow(2);
339 } else {
340 led_lightshow(10);
341 }
342 }
343
344 if (ha_flag_vars_ss.runner) { // timer or trigger from mqtt
345 comedi_push_mqtt(); // send json formatted data to the mqtt server
346 ha_flag_vars_ss.runner = false;
347 }
348 }
349
350 }
351 return 0;
352}
