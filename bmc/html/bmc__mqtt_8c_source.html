590int32_t msgarrvd(void *context, char *topicName, int topicLen, MQTTClient_message *message)
591{
592 int32_t i, ret = 1;
593 const char* payloadptr;
594 char buffer[MBMQTT];
595 struct ha_flag_type *ha_flag = context;
596
597 E.mqtt_count++;
598 // bug-out if no context variables passed to callback
599 if (context == NULL) {
600 ret = -1;
601 goto null_exit;
602 }
603
604#ifdef DEBUG_REC
605 fprintf(fout, "Message arrived\n");
606#endif
607 /*
608 * move the received message into a processing holding buffer
609 */
610 payloadptr = message->payload;
611 if (message->payloadlen > MBMQTT - 1) {
612 fprintf(fout, "%s Error: The received MQTT message is longer than the local message buffer %d\n", log_time(false), MBMQTT);
613 ret = -1;
614 ha_flag->rec_ok = false;
615 E.comedi = false;
616 goto null_exit;
617 }
618 for (i = 0; i < message->payloadlen; i++) {
619 buffer[i] = *payloadptr++;
620 }
621 buffer[i] = 0; // make a null terminated C string
622
623 // parse the JSON data in the holding buffer
624 cJSON *json = cJSON_ParseWithLength(buffer, message->payloadlen);
625 if (json == NULL) {
626 const char *error_ptr = cJSON_GetErrorPtr();
627 if (error_ptr != NULL) {
628 fprintf(fout, "%s Error: %s NULL cJSON pointer\n", log_time(false), error_ptr);
629 }
630 ret = -1;
631 ha_flag->rec_ok = false;
632 E.comedi = false;
633 goto error_exit;
634 }
635
636 /*
637 * MQTT messages for COMEDI
638 */
639#ifdef DEBUG_REC
640 fprintf(fout, "COMEDI MQTT data\r\n");
641#endif
642 cJSON *data_result = json;
643
644 data_result = cJSON_GetObjectItemCaseSensitive(json, "Comedi_Request");
645
646 if (cJSON_IsString(data_result) && (data_result->valuestring != NULL)) {
647 fprintf(fout, "%s Comedi Trigger from MQTT server, Topic %s %s\n", log_time(false), topicName, data_result->valuestring);
648 fflush(fout);
649 ret = true;
650 }
651 E.comedi = true;
652
653 // done with processing MQTT async message, set state flags
654 ha_flag->receivedtoken = true;
655 ha_flag->rec_ok = true;
656 ha_flag_vars_ss.runner = true; // send data in response to received message of any type
657 /*
658 * exit and delete/free resources. In steps depending of possible error conditions
659 */
660error_exit:
661 // delete the JSON object
662 cJSON_Delete(json);
663null_exit:
664 // free the MQTT objects
665 MQTTClient_freeMessage(&message);
666 MQTTClient_free(topicName);
667 fflush(fout);
668 return ret;
669}
